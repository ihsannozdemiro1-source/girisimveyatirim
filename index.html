/* ========================= OTP DOĞRULAMA ========================= */ async function verifyAuthCode() { const email = document.getElementById('authEmail').value.trim(); const token = document.getElementById('otpCode').value.trim(); if (!token || token.length !== 6) { showToast( "Hatalı Kod", "Lütfen 6 haneli doğrulama kodunu girin." ); return; } try { const { data, error } = await sbClient.auth.verifyOtp({ email: email, token: token, type: 'email' }); if (error) { console.error(error); showToast( "Doğrulama Başarısız", error.message ); return; } if (data && data.user) { currentUser = data.user; trackRegisteredUser(data.user.email); updateUserUI(data.user); closeAuthModal(); showToast( "Hesap Doğrulandı", "E-posta adresiniz başarıyla doğrulandı." ); } } catch (error) { console.error(error); showToast( "Hata", "Kod doğrulanırken bir hata oluştu." ); } } /* ========================= KODU TEKRAR GÖNDER ========================= */ async function resendVerificationCode() { const email = document.getElementById('authEmail').value.trim(); if (!email) { showToast( "Hata", "E-posta adresi bulunamadı." ); return; } try { const { error } = await sbClient.auth.resend({ type: 'signup', email: email }); if (error) { console.error(error); showToast( "Kod Gönderilemedi", error.message ); return; } showToast( "Kod Tekrar Gönderildi", "Yeni doğrulama kodu e-posta adresinize gönderildi." ); } catch (error) { console.error(error); showToast( "Hata", "Kod gönderilirken hata oluştu." ); } } /* ========================= OTP'DEN GERİ DÖN ========================= */ function backToAuthForm() { document.getElementById('otpStep').style.display = 'none'; document.getElementById('mainAuthForm').style.display = 'block'; document.getElementById('googleAuthBox').style.display = 'block'; document.getElementById('orDivider').style.display = 'flex'; document.getElementById('authTabHeader').style.display = 'flex'; document.getElementById('authModalTitle').style.display = 'block'; document.getElementById('otpCode').value = ''; }
